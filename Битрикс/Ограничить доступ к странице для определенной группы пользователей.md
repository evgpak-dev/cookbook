Открыть нужный файл страницы (например, /secret.php) и сразу после подключения header.php вставить проверку:

```php
use Bitrix\Main\Application;
use Bitrix\Main\Engine\CurrentUser;

$context = Application::getInstance()->getContext();
$request = $context->getRequest();
$response = $context->getResponse();

$currentUser = CurrentUser::get();
$requiredGroupId = 1; // ID нужной группы пользователей

$hasAccess = in_array($requiredGroupId, $currentUser->getUserGroups());

if (!$hasAccess) {

	// Если пользователь — гость (не авторизован)
	// Перенаправляем на страницу входа с параметром backurl
	if (!$currentUser->getId()) {
		$loginUrl = '/auth/?backurl=' . urlencode($request->getRequestUri());
		LocalRedirect($loginUrl);
		Application::getInstance()->end();
	} else {
		// Пользователь авторизован, но состоит не в той группе
		// Устанавливаем HTTP-статус 403 Forbidden
		$response->setStatus('403 Forbidden');
	}

?>
	<div class="container my-5">
		<div class="alert alert-danger">
			<h4>403 Доступ ограничен</h4>
			<p>У вашей учетной записи недостаточно прав для просмотра данного раздела.</p>
		</div>
	</div>
<?php
	require($_SERVER["DOCUMENT_ROOT"] . "/bitrix/footer.php");
	Application::getInstance()->end();
}
```
